import 'package:flutter/material.dart';
import 'package:google_fonts/google_fonts.dart';
import 'signup_screen.dart';
import 'login_screen.dart';

class WelcomeScreen extends StatelessWidget {
  const WelcomeScreen({super.key});

  @override
  Widget build(BuildContext context) {
    // Gradient gold renkleri
    const goldGradient = LinearGradient(
      colors: [
        Color(0xFFFFD700), // Parlak gold
        Color(0xFFD4AF37), // Orta gold
        Color(0xFFB8860B), // Koyu gold
        Color(0xFFD4AF37), // Orta gold
        Color(0xFFFFD700), // Parlak gold
      ],
      begin: Alignment.topLeft,
      end: Alignment.bottomRight,
    );

    return Scaffold(
      backgroundColor: Colors.black,
      body: SafeArea(
        child: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 40.0),
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              const Spacer(flex: 2),

              // QMatch Logo - Gradient gold
              ShaderMask(
                shaderCallback: (bounds) => goldGradient.createShader(bounds),
                child: Image.asset(
                  'assets/images/logo_white.png',
                  width: 320,
                  height: 220,
                  filterQuality: FilterQuality.high,
                  color: Colors.white,
                ),
              ),

              const SizedBox(height: 20),

              // Slogan - Gradient gold
              ShaderMask(
                shaderCallback: (bounds) => goldGradient.createShader(bounds),
                child: Text(
                  'text text text',
                  style: GoogleFonts.playfairDisplay(
                    color: Colors.white,
                    fontSize: 18,
                    fontWeight: FontWeight.w400,
                    fontStyle: FontStyle.italic,
                    letterSpacing: 1,
                  ),
                ),
              ),

              const Spacer(flex: 2),

              // Sign Up Button - Gradient border
              Container(
                decoration: BoxDecoration(
                  gradient: goldGradient,
                  borderRadius: BorderRadius.circular(16),
                ),
                child: Container(
                  margin: const EdgeInsets.all(2),
                  decoration: BoxDecoration(
                    color: Colors.black,
                    borderRadius: BorderRadius.circular(14),
                  ),
                  child: SizedBox(
                    width: double.infinity,
                    height: 60,
                    child: TextButton(
                      onPressed: () {
                        Navigator.push(
                          context,
                          MaterialPageRoute(
                            builder: (context) => const SignupScreen(),
                          ),
                        );
                      },
                      child: ShaderMask(
                        shaderCallback: (bounds) =>
                            goldGradient.createShader(bounds),
                        child: Text(
                          'Sign Up',
                          style: GoogleFonts.inter(
                            color: Colors.white,
                            fontSize: 18,
                            fontWeight: FontWeight.w600,
                            letterSpacing: 0.5,
                          ),
                        ),
                      ),
                    ),
                  ),
                ),
              ),

              const SizedBox(height: 20),

              // Login Button - Gradient border
              Container(
                decoration: BoxDecoration(
                  gradient: goldGradient,
                  borderRadius: BorderRadius.circular(16),
                ),
                child: Container(
                  margin: const EdgeInsets.all(2),
                  decoration: BoxDecoration(
                    color: Colors.black,
                    borderRadius: BorderRadius.circular(14),
                  ),
                  child: SizedBox(
                    width: double.infinity,
                    height: 60,
                    child: TextButton(
                      onPressed: () {
                        Navigator.push(
                          context,
                          MaterialPageRoute(
                            builder: (context) => const LoginScreen(),
                          ),
                        );
                      },
                      child: ShaderMask(
                        shaderCallback: (bounds) =>
                            goldGradient.createShader(bounds),
                        child: Text(
                          'Login',
                          style: GoogleFonts.inter(
                            color: Colors.white,
                            fontSize: 18,
                            fontWeight: FontWeight.w600,
                            letterSpacing: 0.5,
                          ),
                        ),
                      ),
                    ),
                  ),
                ),
              ),

              const Spacer(flex: 1),
            ],
          ),
        ),
      ),
    );
  }
}
